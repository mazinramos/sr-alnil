/* ============================================================================
   مرجع: صيغ الألغاز الاثنا عشر (انسخي ما تحتاجين إلى puzzle)
   ----------------------------------------------------------------------------
   sequence : { type:"sequence", items:[{id,label}], solution:["id",...] }
   hotspot  : { type:"hotspot", needed:3, targets:[{id,label,x,y,r}] }
   matching : { type:"matching", left:[Item], right:[Item], solution:{leftId:rightId} }
   codelock : { type:"codelock", length:4, symbols:[Item], solution:["id",...] }   // زر التأكيد «افتحي»
   choice   : { type:"choice", question, options:[Item], correct:["id"], multi?:true }
   route    : { type:"route", pick:5, nodes:[{id,x,y,label,detail?}], solution:["id",...] }
   balance  : { type:"balance", target:30, maxItems?:3, weights:[{id,label,value}] }
   tiles    : { type:"tiles", image?:"/art/cN/lX-puzzle.webp", size:[3,3],
                tiles:[{id,rot}], solution:[{id,rot}] }     // ترتيب solution = موضع القصّ من الصورة
   memory   : { type:"memory", pairs:[Item] }
   cipher   : { type:"cipher", key:[{id,label,detail?}], cipherText:["k1","k2","|","k3"],
                answer:["ا","ل","|","م"] }   // "|" فاصل كلمات؛ detail = الحرف المقابل ويجب أن يطابق answer
   deduction: { type:"deduction", rows:[Item], cols:[Item], solution:{rowId:colId} }
   jigsaw   : { type:"jigsaw", image:"/art/ui/map-full.webp#cell=N", pieces:2..15 }
   Item = { id, label, detail?, icon? }   // icon: معرّف أيقونة أو مسار يبدأ بـ /
   ============================================================================ */
